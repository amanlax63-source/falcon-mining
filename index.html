import json import logging from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup, WebAppInfo from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, filters, ContextTypes # Logging ማስተካከያ logging.basicConfig( format='%(asctime)s - %(name)s - %(levelname)s - %(message)s', level=logging.INFO ) # ---------------- CONFIGURATION ---------------- BOT_TOKEN = "YOUR_BOT_TOKEN_HERE" # ከ BotFather ያገኘኸው Token MINI_APP_URL = "https://your-domain.com/index.html" # የ Mini App-ህ URL ADMIN_ID = 123456789 # የ Admin Telegram Chat ID # ---------------- HANDLERS ---------------- async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user # ሪፈራል መፈተሽ (ከ /start በኋላ ያለውን ID መውሰድ) args = context.args referrer_id = args[0] if args else None if referrer_id: print(f"User {user.id} was referred by {referrer_id}") # Mini App የሚከፍት አዝራር (Keyboard Button) keyboard = [ [ InlineKeyboardButton( text="🚀 Mini App ክፈት", web_app=WebAppInfo(url=MINI_APP_URL) ) ] ] reply_markup = InlineKeyboardMarkup(keyboard) welcome_text = ( f"ሰላም {user.first_name}! 👋\n\n" "እንኳን ወደ እኛ **Earning Mini App** በሰላም መጡ! 🤑\n\n" "ከታች ያለውን **'Mini App ክፈት'** የሚለውን በመጫን ባላንስዎን ማየት፣ " "የሪፈራል ሊንክዎን መውሰድ እና ብር ማውጣት ይችላሉ።" ) await update.message.reply_text(welcome_text, reply_markup=reply_markup, parse_mode="Markdown") # ከ Mini App የተላከውን Data መቀበያ (tg.sendData) async def web_app_data_handler(update: Update, context: ContextTypes.DEFAULT_TYPE): data_str = update.effective_message.web_app_data.data data = json.loads(data_str) if data.get("action") == "withdraw": user_id = data.get("userId") username = data.get("username") amount = data.get("amount") account = data.get("account") # ለተጠቃሚው ማረጋገጫ መላክ await update.message.reply_text( f"✅ **የማውጫ ጥያቄዎ ደርሶናል!**\n\n" f"💰 መጠን: {amount} ETB\n" f"💳 ሂሳብ ቁጥር: {account}\n\n" f"ጥያቄዎ በአጭር ጊዜ ውስጥ ተመርምሮ ይላክልዎታል።", parse_mode="Markdown" ) # ለ Admin ማሳወቂያ መላክ admin_message = ( f"🚨 **አዲስ የማውጫ ጥያቄ (Withdrawal)**\n\n" f"👤 ተጠቃሚ: @{username} (ID: `{user_id}`)\n" f"💰 መጠን: **{amount} ETB**\n" f"💳 ሂሳብ/ስልክ ቁጥር: `{account}`" ) await context.bot.send_message(chat_id=ADMIN_ID, text=admin_message, parse_mode="Markdown") # ---------------- MAIN FUNCTION ---------------- def main(): app = ApplicationBuilder().token(BOT_TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(MessageHandler(filters.StatusUpdate.WEB_APP_DATA, web_app_data_handler)) print("🤖 ቦቱ በስኬት ስራ ጀምሯል...") app.run_polling() if __name__ == "__main__": main()
